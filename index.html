CREATE PROCEDURE GetOptimizedUserData(IN p_userId INT)
BEGIN
  -- Automated 3NF relational retrieval with indexed scan
  SELECT u.id, u.username, u.email, r.role_name
  FROM users_auth u
  INNER JOIN user_roles r ON u.role_id = r.id
  WHERE u.id = p_userId AND u.is_active = 1;
END;
